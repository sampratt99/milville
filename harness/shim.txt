/* ============================================================================
   shim.txt — the offline stand-in for the browser.

   Prepend this to a Pattern-B harness (see docs/14_VALIDATION_HARNESSES.md).
   It installs a strict-mode-safe DOM, a canvas 2D context that counts its own
   save/restore depth, and a THREE r128 stub wide enough that the whole 47k-line
   module scope of index.html executes top to bottom without throwing.

   Nothing here fires on its own. requestAnimationFrame is a no-op (the file ends
   with `requestAnimationFrame(render)` at top level) and setTimeout only records
   the callback — flush it deliberately with __shim.flushTimers() if a test wants
   deferred work to happen. That is what keeps a harness deterministic.

   Known limits, so nobody tests the shim instead of the game:
     - no layout: getBoundingClientRect is a fixed 800x600
     - no WebGL: nothing is rasterised, so appearance is unprovable here
     - Object3D.visible and .rotation ARE readable (unlike the older shim), but
       they only reflect what the game set, never what a renderer would show
   ========================================================================== */
'use strict';

/* Node already owns some of these names (navigator and fetch are getter-only in
   Node 22+), so every global goes in by defineProperty rather than assignment. */
function def(name, value){
  Object.defineProperty(globalThis, name, {value, writable:true, configurable:true, enumerable:false});
  return value;
}

const __shim = def('__shim', {
  timers: [],           /* every setTimeout the game asked for, in order */
  intervals: [],
  raf: [],
  msgs: [],             /* whatever msg() writes, if a harness wires it up */
  flushTimers,
  reset(){ this.timers.length=0; this.intervals.length=0; this.raf.length=0; },
});

/* run pending timers once, in scheduled order; returns how many fired */
function flushTimers(){
  const due = __shim.timers.splice(0, __shim.timers.length);
  due.sort((a,b)=>a.ms-b.ms);
  let n=0;
  for(const t of due){ if(t.cancelled) continue; try{ t.fn(...t.args); n++; }catch(e){ t.error=e; } }
  return n;
}

/* ------------------------------------------------------------------ timers */
let __timerId = 1;
def('setTimeout', function(fn, ms, ...args){
  const id = __timerId++;
  __shim.timers.push({id, fn, ms: ms|0, args, cancelled:false});
  return id;
});
def('clearTimeout', function(id){
  const t = __shim.timers.find(t=>t.id===id); if(t) t.cancelled = true;
});
def('setInterval', function(fn, ms, ...args){
  const id = __timerId++;
  __shim.intervals.push({id, fn, ms: ms|0, args});
  return id;
});
def('clearInterval', function(id){
  const i = __shim.intervals.findIndex(t=>t.id===id); if(i>=0) __shim.intervals.splice(i,1);
});
def('requestAnimationFrame', function(fn){ __shim.raf.push(fn); return __shim.raf.length; });
def('cancelAnimationFrame', function(){});

/* Every drawing call is a no-op, but save()/restore() keep a real depth counter
   so `iconaudit`-style tests can prove a drawer balances its own canvas. */
function makeCtx2D(canvas){
  const ctx = {
    canvas,
    __depth: 0,          /* save() minus restore() — must return to 0 */
    __maxDepth: 0,
    __calls: 0,
  };
  const props = {
    fillStyle:'#000', strokeStyle:'#000', lineWidth:1, lineCap:'butt', lineJoin:'miter',
    miterLimit:10, globalAlpha:1, globalCompositeOperation:'source-over',
    font:'10px sans-serif', textAlign:'start', textBaseline:'alphabetic', direction:'ltr',
    shadowBlur:0, shadowColor:'rgba(0,0,0,0)', shadowOffsetX:0, shadowOffsetY:0,
    imageSmoothingEnabled:true, imageSmoothingQuality:'low', filter:'none', lineDashOffset:0,
  };
  for(const k in props) ctx[k] = props[k];

  const noop = ['beginPath','closePath','moveTo','lineTo','bezierCurveTo','quadraticCurveTo',
    'arc','arcTo','ellipse','rect','roundRect','fill','stroke','clip','fillRect','strokeRect',
    'clearRect','fillText','strokeText','drawImage','setLineDash','scale','rotate','translate',
    'transform','setTransform','resetTransform','putImageData','drawFocusIfNeeded','clip'];
  for(const m of noop) ctx[m] = function(){ ctx.__calls++; };

  ctx.save = function(){ ctx.__calls++; ctx.__depth++; if(ctx.__depth>ctx.__maxDepth) ctx.__maxDepth=ctx.__depth; };
  ctx.restore = function(){ ctx.__calls++; ctx.__depth--; };
  ctx.measureText = function(t){
    const w = (t==null?0:String(t).length) * 6;
    return {width:w, actualBoundingBoxLeft:0, actualBoundingBoxRight:w,
            actualBoundingBoxAscent:8, actualBoundingBoxDescent:2,
            fontBoundingBoxAscent:8, fontBoundingBoxDescent:2};
  };
  ctx.createLinearGradient = ctx.createRadialGradient = ctx.createConicGradient =
    function(){ return {addColorStop(){}}; };
  ctx.createPattern = function(){ return {setTransform(){}}; };
  ctx.getImageData = function(x,y,w,h){
    w = Math.max(1, w|0); h = Math.max(1, h|0);
    return {width:w, height:h, data:new Uint8ClampedArray(w*h*4)};
  };
  ctx.createImageData = ctx.getImageData;
  ctx.getLineDash = function(){ return []; };
  ctx.isPointInPath = ctx.isPointInStroke = function(){ return false; };
  ctx.getTransform = function(){ return {a:1,b:0,c:0,d:1,e:0,f:0}; };
  return ctx;
}

class ShimClassList {
  constructor(){ this._s = new Set(); }
  add(...c){ for(const x of c) if(x) this._s.add(x); }
  remove(...c){ for(const x of c) this._s.delete(x); }
  toggle(c, force){
    if(force===true){ this._s.add(c); return true; }
    if(force===false){ this._s.delete(c); return false; }
    if(this._s.has(c)){ this._s.delete(c); return false; }
    this._s.add(c); return true;
  }
  contains(c){ return this._s.has(c); }
  item(i){ return [...this._s][i] ?? null; }
  get length(){ return this._s.size; }
  toString(){ return [...this._s].join(' '); }
  [Symbol.iterator](){ return this._s[Symbol.iterator](); }
}

/* Not a renderer — a bookkeeper. Scene-graph structure, transforms, visibility
   and material/geometry parameters are all real and readable; nothing is ever
   rasterised. Enough for houseBuildInterior() and every model builder to run. */
class V3 {
  constructor(x=0, y=0, z=0){ this.x=x; this.y=y; this.z=z; }
  set(x,y,z){ this.x=x; this.y=y; this.z=z; return this; }
  setScalar(s){ this.x=this.y=this.z=s; return this; }
  setX(v){ this.x=v; return this; } setY(v){ this.y=v; return this; } setZ(v){ this.z=v; return this; }
  copy(v){ this.x=v.x||0; this.y=v.y||0; this.z=v.z||0; return this; }
  clone(){ return new V3(this.x, this.y, this.z); }
  add(v){ this.x+=v.x; this.y+=v.y; this.z+=v.z; return this; }
  addScalar(s){ this.x+=s; this.y+=s; this.z+=s; return this; }
  addVectors(a,b){ this.x=a.x+b.x; this.y=a.y+b.y; this.z=a.z+b.z; return this; }
  addScaledVector(v,s){ this.x+=v.x*s; this.y+=v.y*s; this.z+=v.z*s; return this; }
  sub(v){ this.x-=v.x; this.y-=v.y; this.z-=v.z; return this; }
  subVectors(a,b){ this.x=a.x-b.x; this.y=a.y-b.y; this.z=a.z-b.z; return this; }
  multiply(v){ this.x*=v.x; this.y*=v.y; this.z*=v.z; return this; }
  multiplyScalar(s){ this.x*=s; this.y*=s; this.z*=s; return this; }
  divideScalar(s){ return this.multiplyScalar(s===0?0:1/s); }
  negate(){ this.x=-this.x; this.y=-this.y; this.z=-this.z; return this; }
  dot(v){ return this.x*v.x + this.y*v.y + this.z*v.z; }
  cross(v){ const {x,y,z}=this;
    this.x=y*v.z-z*v.y; this.y=z*v.x-x*v.z; this.z=x*v.y-y*v.x; return this; }
  crossVectors(a,b){ return this.copy(a).cross(b); }
  length(){ return Math.hypot(this.x, this.y, this.z); }
  lengthSq(){ return this.x**2 + this.y**2 + this.z**2; }
  normalize(){ return this.divideScalar(this.length()||1); }
  setLength(l){ return this.normalize().multiplyScalar(l); }
  distanceTo(v){ return Math.hypot(this.x-v.x, this.y-v.y, this.z-v.z); }
  distanceToSquared(v){ return (this.x-v.x)**2 + (this.y-v.y)**2 + (this.z-v.z)**2; }
  lerp(v,t){ this.x+=(v.x-this.x)*t; this.y+=(v.y-this.y)*t; this.z+=(v.z-this.z)*t; return this; }
  applyQuaternion(){ return this; }
  applyMatrix4(){ return this; }
  applyMatrix3(){ return this; }
  applyEuler(){ return this; }
  applyAxisAngle(){ return this; }
  transformDirection(){ return this; }
  clampScalar(a, b){ this.x = Math.max(a, Math.min(b, this.x));

  clampLength(a, b){ const l = this.length() || 1;

  min(v){ this.x = Math.min(this.x, v.x); this.y = Math.min(this.y, v.y);

  floor(){ this.x = Math.floor(this.x); this.y = Math.floor(this.y);

class Color {
  constructor(r, g, b){ this.r=1; this.g=1; this.b=1; if(r!==undefined) this.set(r, g, b); }
  set(r, g, b){
    if(g !== undefined && b !== undefined){ this.r=r; this.g=g; this.b=b; return this; }
    if(typeof r === 'number'){ return this.setHex(r); }
    if(r instanceof Color) return this.copy(r);
    if(typeof r === 'string'){
      const m = /^#?([0-9a-f]{6})$/i.exec(r.trim());
      if(m) return this.setHex(parseInt(m[1], 16));
    }
    return this;
  }
  setHex(h){ h = Math.floor(h);
    this.r = ((h>>16)&255)/255; this.g = ((h>>8)&255)/255; this.b = (h&255)/255; return this; }
  setRGB(r,g,b){ this.r=r; this.g=g; this.b=b; return this; }
  setScalar(s){ this.r=this.g=this.b=s; return this; }
  setHSL(h,s,l){ this.r=this.g=this.b=l; return this; }   /* value-shaped, not colour-accurate */
  setStyle(s){ return this.set(s); }
  copy(c){ this.r=c.r; this.g=c.g; this.b=c.b; return this; }
  clone(){ return new Color(this.r, this.g, this.b); }
  getHex(){ return (Math.round(this.r*255)<<16) ^ (Math.round(this.g*255)<<8) ^ Math.round(this.b*255); }
  getHexString(){ return this.getHex().toString(16).padStart(6, '0'); }
  getStyle(){ return '#' + this.getHexString(); }
  getHSL(t){ t = t || {}; t.h=0; t.s=0; t.l=(this.r+this.g+this.b)/3; return t; }
  offsetHSL(){ return this; }
  add(c){ this.r+=c.r; this.g+=c.g; this.b+=c.b; return this; }
  multiply(c){ this.r*=c.r; this.g*=c.g; this.b*=c.b; return this; }
  multiplyScalar(s){ this.r*=s; this.g*=s; this.b*=s; return this; }
  lerp(c,t){ this.r+=(c.r-this.r)*t; this.g+=(c.g-this.g)*t; this.b+=(c.b-this.b)*t; return this; }
  convertSRGBToLinear(){ return this; }
  convertLinearToSRGB(){ return this; }
}

}

/* Every primitive carries a small but structurally REAL vertex buffer. bake()
   walks geo.attributes.position.array / .normal.array and pushes them into a
   merged mesh, so these cannot be empty or mismatched — but nothing offline
   reads their values, only their shape. 12 vertices keeps a 47k-tile forest
   bake cheap while staying a valid triangle list. */
const __GEO_VERTS = 12;
class Geometry {
  constructor(type, params){
    this.type = type;
    this.uuid = 'geo-' + (++__objSeq);
    this.parameters = params || {};
    this.attributes = {};
    this.index = null;
    if(type !== 'BufferGeometry'){
      const n = __GEO_VERTS * 3;
      const pos = new Float32Array(n);
      const nor = new Float32Array(n);
      for(let i = 0; i < n; i += 3){ nor[i] = 0; nor[i+1] = 1; nor[i+2] = 0; }
      this.attributes.position = new Float32BufferAttribute(pos, 3);
      this.attributes.normal   = new Float32BufferAttribute(nor, 3);
      this.attributes.uv       = new Float32BufferAttribute(new Float32Array(__GEO_VERTS * 2), 2);
    }
    this.groups = [];
    this.boundingBox = null;
    this.boundingSphere = null;
    /* the game reads back BoxGeometry dimensions in a few places */
    const p = this.parameters;
    if(p.width !== undefined) this.width = p.width;
    if(p.height !== undefined) this.height = p.height;
    if(p.depth !== undefined) this.depth = p.depth;
  }
  setAttribute(n, a){ this.attributes[n] = a; return this; }
  getAttribute(n){ return this.attributes[n]; }
  deleteAttribute(n){ delete this.attributes[n]; return this; }
  setIndex(i){ this.index = i; return this; }
  setFromPoints(pts){ this.__points = pts; return this; }
  addGroup(s, c, m){ this.groups.push({start:s, count:c, materialIndex:m}); return this; }
  clearGroups(){ this.groups.length = 0; return this; }
  translate(x, y, z){ this.__t = {x, y, z}; return this; }
  rotateX(a){ this.__rx = (this.__rx||0) + a; return this; }
  rotateY(a){ this.__ry = (this.__ry||0) + a; return this; }
  rotateZ(a){ this.__rz = (this.__rz||0) + a; return this; }
  scale(x, y, z){ this.__s = {x, y, z}; return this; }
  center(){ return this; }
  lookAt(){ return this; }
  applyMatrix4(){ return this; }
  computeVertexNormals(){ return this; }
  computeBoundingBox(){ this.boundingBox = new Box3(); return this; }
  computeBoundingSphere(){ this.boundingSphere = {center:new V3(), radius:1}; return this; }
  normalizeNormals(){ return this; }
  toNonIndexed(){ return this; }
  clone(){
    const g = new Geometry('BufferGeometry', this.parameters);
    g.type = this.type;
    for(const k in this.attributes){
      const a = this.attributes[k];
      g.attributes[k] = new Float32BufferAttribute(Float32Array.from(a.array), a.itemSize);
    }
    g.index = this.index;
    return g;
  }
  copy(){ return this; }
  dispose(){}
}

  };
}
